pragma solidity ^0.8.20;

contract StudentData {

    // Structure
    struct Student {
        uint rollNo;
        string name;
        uint marks;
    }

    // Array of students
    Student[] public students;

    // Event for fallback
    event FallbackCalled(address sender);

    // Add student
    function addStudent(
        uint _rollNo,
        string memory _name,
        uint _marks
    ) public {

        students.push(
            Student(_rollNo, _name, _marks)
        );
    }

    // Get student data
    function getStudent(uint index)
        public
        view
        returns (
            uint,
            string memory,
            uint
        )
    {
        Student memory s = students[index];

        return (
            s.rollNo,
            s.name,
            s.marks
        );
    }

    // Get total number of students
    function getStudentCount()
        public
        view
        returns (uint)
    {
        return students.length;
    }

    // Fallback function
    fallback() external {
        emit FallbackCalled(msg.sender);
    }
}
