def safe(b, r, c, n):
    for i in range(n):
        if b[i][c] == 1:
            return False
        for j in range(n):
            if b[i][j] == 1 and abs(r-i) == abs(c-j):
                return False
    return True

def solve(b, r, n, qr):
    if r == n:
        return True
    if r == qr:
        return solve(b, r+1, n, qr)

    for c in range(n):
        if safe(b, r, c, n):
            b[r][c] = 1
            if solve(b, r+1, n, qr):
                return True
            b[r][c] = 0
    return False

n = int(input("Enter N: "))
b = [[0]*n for _ in range(n)]

qr = int(input("Enter first Queen row: ")) - 1
qc = int(input("Enter first Queen column: ")) - 1

if 0 <= qr < n and 0 <= qc < n:
    b[qr][qc] = 1
    if solve(b, 0, n, qr):
        print("Final N-Queens Matrix:")
        for row in b:
            print(*row)
    else:
        print("No solution exists")
else:
    print("Invalid position")
